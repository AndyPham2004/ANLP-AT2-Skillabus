# app/enrol.py
"""Can this student enrol in that subject, and when does it run?

Reads data/processed/prerequisites.csv (one row per rule item, from
data-eda/src/scrape_requisites.py) and offerings.csv. A rule is a boolean expression over
item ids, e.g. "((1 OR 1a) AND 2)". An item is satisfied when:
  admission  the text names the student's home course (MDSI, C04372)
  academic   a 5-digit subject code the student has taken; credit-point conditions count as
             not met (we cannot see the transcript)
  other      counts as met (cannot be checked here)
Everything here is pure Python so it is unit-tested without Streamlit.
"""

from __future__ import annotations

import csv
import re
from functools import lru_cache
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
PREREQ_CSV = ROOT / "data" / "processed" / "prerequisites.csv"
OFFER_CSV = ROOT / "data" / "processed" / "offerings.csv"
PROFILES_CSV = ROOT / "data" / "processed" / "subject_profiles.csv"
HOME_COURSE = "C04372"
TOKEN = re.compile(r"\(|\)|AND|OR|\d+[a-z]?")


def _subject_courses() -> dict[str, str]:
    """subject_code -> pipe-joined course codes, from the profiles (empty when the file is absent)."""
    if not PROFILES_CSV.exists():
        return {}
    with open(PROFILES_CSV, encoding="utf-8-sig", newline="") as f:
        return {row["subject_code"]: row.get("course", "") for row in csv.DictReader(f)}


SUBJECT_COURSES = _subject_courses()
KNOWN_SUBJECTS = frozenset(SUBJECT_COURSES)


def course_of(code: str) -> str:
    return SUBJECT_COURSES.get(code, "")


def is_bachelor(code: str) -> bool:
    """Every course the subject sits in is an undergraduate one (UTS codes C10xxx)."""
    courses = [c for c in course_of(code).split("|") if c]
    return bool(courses) and all(c.startswith("C10") for c in courses)


@lru_cache(maxsize=1)
def _load() -> tuple[dict[str, dict], dict[str, list[str]]]:
    rules: dict[str, dict] = {}
    if PREREQ_CSV.exists():
        with open(PREREQ_CSV, encoding="utf-8-sig", newline="") as f:
            for row in csv.DictReader(f):
                r = rules.setdefault(row["subject_code"], {"rule": row["rule"] or "", "items": []})
                if row["item"]:
                    r["items"].append({"item": row["item"], "type": row["type"],
                                       "code": row["requisite_code"] or "", "text": row["requisite_text"] or ""})
    sessions: dict[str, list[str]] = {}
    if OFFER_CSV.exists():
        with open(OFFER_CSV, encoding="utf-8-sig", newline="") as f:
            for row in csv.DictReader(f):
                s = sessions.setdefault(row["subject_code"], [])
                if row["teaching_period"] not in s:
                    s.append(row["teaching_period"])
    return rules, sessions


def _item_met(item: dict, taken: set[str], home_course: str) -> bool:
    if item["type"] == "admission":
        return home_course in item["text"]
    if item["type"] == "academic":
        return bool(item["code"]) and item["code"] in taken
    return True


def eval_rule(rule: str, items: list[dict], taken: set[str], home_course: str = HOME_COURSE) -> bool | None:
    """Evaluate "((1 OR 1a) AND 2)" against the items; None when there is no rule."""
    if not rule.strip():
        return None
    met = {it["item"]: _item_met(it, taken, home_course) for it in items}
    tokens = TOKEN.findall(rule)
    pos = 0

    def parse_or():
        nonlocal pos
        value = parse_and()
        while pos < len(tokens) and tokens[pos] == "OR":
            pos += 1
            value = parse_and() or value
        return value

    def parse_and():
        nonlocal pos
        value = parse_atom()
        while pos < len(tokens) and tokens[pos] == "AND":
            pos += 1
            value = parse_atom() and value
        return value

    def parse_atom():
        nonlocal pos
        tok = tokens[pos]
        pos += 1
        if tok == "(":
            value = parse_or()
            pos += 1          # the closing ")"
            return value
        return met.get(tok, False)

    try:
        return bool(parse_or())
    except IndexError:        # a malformed rule: say "unknown" rather than block the subject
        return None


def enrolment(code: str, taken: set[str], home_course: str = HOME_COURSE) -> dict:
    rules, sessions = _load()
    r = rules.get(code, {"rule": "", "items": []})
    needed = [it["code"] for it in r["items"] if it["type"] == "academic" and it["code"]]
    missing = [c for c in needed if c not in taken]
    return {
        "rule": r["rule"],
        "can_enrol": eval_rule(r["rule"], r["items"], taken, home_course),
        "missing": missing,
        "outside": [c for c in missing if c not in KNOWN_SUBJECTS],
        "sessions": sessions.get(code, []),
    }
```

from __future__ import annotations

import io


def read_upload(name: str, data: bytes) -> str:
    if name.lower().endswith(".pdf"):
        from pypdf import PdfReader
        try:
            return "\n".join((page.extract_text() or "") for page in PdfReader(io.BytesIO(data)).pages).strip()
        except Exception:
            return ""
    return data.decode("utf-8", errors="ignore")

# app/render.py
"""Pure functions behind the results page: no Streamlit here, so they are unit-tested.

highlight_ad   the ad text with every skill mention wrapped in <mark>, coloured by status
coverage_pct   how much of the ad's demand-weighted skills the resume already covers
verdict        the one-line summary at the top of the results
what_if        the numbers after the student ticks the subjects they will take
schedule       which upcoming session each path subject can be taken in
"""

from __future__ import annotations

import html
import re

TIER = {"llm": "GPT-4o-mini", "ner": "spaCy", "dictionary": "dictionary", None: "dictionary"}


def highlight_ad(ad_text: str, skills: list[dict], pattern_for) -> str:
    """HTML of the ad with each skill occurrence in <mark class="sk-<status>" title="...">."""
    spans: list[tuple[int, int, dict]] = []
    for s in skills:
        for m in pattern_for(s["skill_label"]).finditer(ad_text):
            spans.append((m.start(), m.end(), s))
    spans.sort(key=lambda t: (t[0], -(t[1] - t[0])))
    out, pos, last_end = [], 0, -1
    for start, end, s in spans:
        if start < last_end:            # overlapping with a longer, earlier match: skip
            continue
        out.append(html.escape(ad_text[pos:start]))
        title = f"{s['skill_label']} · {s['status']} · found by {TIER.get(s.get('found_by'), 'dictionary')}"
        out.append(f'<mark class="sk-{s["status"]}" title="{html.escape(title)}">{html.escape(ad_text[start:end])}</mark>')
        pos, last_end = end, end
    out.append(html.escape(ad_text[pos:]))
    return "".join(out).replace("\n", "<br>")
```

```python
SESSIONS = ("Autumn 2027", "Spring 2027", "Autumn 2028", "Spring 2028")
CREDIT = {"have": 1.0, "partial": 0.5, "missing": 0.0}


def _w(s: dict) -> float:
    return float(s.get("demand_pct") or 0.0) + 1.0


def lc(label: str) -> str:
    return label[:1].lower() + label[1:] if len(label) > 1 and label[1].islower() else label


def coverage_pct(skills: list[dict]) -> float:
    total = sum(_w(s) for s in skills)
    return 100.0 * sum(_w(s) * CREDIT[s["status"]] for s in skills) / total if total else 0.0


def verdict(result: dict, target: str) -> str:
    skills = result["skills"]
    gaps = sorted((s for s in skills if s["status"] != "have"), key=lambda s: -_w(s))
    lp = result.get("learning_path") or {}
    parts = [f"You cover {coverage_pct(skills):.0f}% of what {target} asks for."]
    if gaps:
        parts.append("Biggest gaps: " + ", ".join(lc(g["skill_label"]) for g in gaps[:3]) + ".")
    n = len(lp.get("subjects") or [])
    if n:
        parts.append(f"{n} subject{'s' if n > 1 else ''} close {lp['gaps_closed']} of {lp['gaps_total']} gaps.")
    return " ".join(parts)


def what_if(result: dict, chosen: set[str]) -> dict:
    gaps = [s for s in result["skills"] if s["status"] != "have"]
    recs = list(result.get("recommendations") or []) + list(result.get("outside_recommendations") or [])
    covered = {label for r in recs if r["subject_code"] in chosen for label in r["covers"]}
    closed = [g["skill_label"] for g in gaps if g["skill_label"] in covered]
    remaining = [g["skill_label"] for g in gaps if g["skill_label"] not in covered]
    total = sum(_w(g) for g in gaps)
    pct = round(100.0 * sum(_w(g) for g in gaps if g["skill_label"] in covered) / total, 1) if total else 0.0
    return {"closed": closed, "remaining": remaining, "pct": pct}


def schedule(path_codes: list[str], enrol_fn, taken: set[str], sessions: tuple[str, ...] = SESSIONS) -> list[tuple[str, list[str]]]:
    """Place each subject in the first upcoming session it runs in once its prerequisites are done."""
    done = set(taken)
    pending = list(path_codes)
    out: list[tuple[str, list[str]]] = []
    for session in sessions:
        season = session.split()[0]
        placed = []
        for code in pending:
            info = enrol_fn(code, done)
            runs = any(p.startswith(season) for p in info["sessions"]) or not info["sessions"]
            if runs and not [m for m in info["missing"] if m not in done]:
                placed.append(code)
        if placed:
            out.append((session, placed))
            done |= set(placed)
            pending = [c for c in pending if c not in placed]
    if pending:
        out.append(("Not yet possible", pending))
    return out
```

EMAIL = re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+")
_TITLE_WORDS = {"data", "scientist", "engineer", "analyst", "resume", "cv", "curriculum", "vitae", "profile", "summary", "developer", "manager"}


def _mask_name_line(text: str) -> str:
    lines = text.split("\n")
    for i, line in enumerate(lines):
        if not line.strip():
            continue
        words = line.strip().split()
        if 2 <= len(words) <= 4 and all(w[:1].isupper() for w in words) and not any(ch.isdigit() for ch in line) \
                and not ({w.lower().strip(".,") for w in words} & _TITLE_WORDS):
            lines[i] = "[NAME]"
        break
    return "\n".join(lines)
